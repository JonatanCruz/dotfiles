# Casos de redirect-code-search-to-mcp.sh — el gate que manda al índice.
#
# Los comandos corren con cwd en un repo armado con fixtures/repo-casos:
#   pkg/a/src/*.ts · pkg/b/lib/*.py · src/app.ts · src/README.md
#   deploy/*.tf · deploy/dev/*.yml (sólo YAML) · db/*.sql · docs/*.md (sólo md)
# El gate MIDE si un directorio contiene código, así que estas rutas existen.
#
# ── POR QUÉ EXISTE ESTE ARCHIVO ────────────────────────────────────────────
#
# Este gate NO tiene override. Un gate sin escape hatch tiene que ser exacto:
# cada falso positivo es trabajo que no se puede hacer de ninguna forma, y
# entrena a rodearlo por vías peores. Y no tiene nada escrito para ningún repo:
# si un caso de acá sólo pasa con un nombre de directorio concreto, el gate
# volvió a acoplarse a un proyecto.
#
# Formato:  <exit esperado> <TAB> <descripción> <TAB> <comando>

# ── DEBE BLOQUEAR (exit 2): búsqueda recursiva sobre código ───────────────

2	grep -rn sobre un directorio con .ts	grep -rn "handler" pkg/a/src/
2	rg recursivo sobre un directorio con código	rg -n "handler" pkg/
2	grep -r sobre src/	grep -r "export" src/
2	grep con comodín sobre código	grep -n "x" pkg/a/src/*.ts
2	grep --include sobre el repo	grep -n --include=*.ts "x" .
2	directorio con .tf	grep -rn port deploy/

# ── DEBE PASAR (exit 0): lo que el mensaje del gate PROMETE permitir ──────
#
# "un archivo concreto, sin recursión ni comodines"

0	UN archivo .ts concreto	grep -n handler pkg/a/src/index.ts
0	UN archivo .tf concreto	grep -oE "[0-9.]+" deploy/main.tf
0	UN archivo .sql concreto	grep -n CREATE db/0001_init.sql
0	UN archivo .astro concreto	grep -n props pkg/a/src/pages/index.astro
0	UN archivo, aunque lleve -r	grep -rn handler pkg/a/src/index.ts

# "texto que no sea código: logs, JSON, YAML, configs, salida de CLIs"

0	log	grep -n error /var/log/syslog
0	JSON	grep -n name package.json
0	YAML	grep -n x docker-compose.yml
0	salida de CLI por pipe	docker compose ps | grep healthy
0	markdown	grep -rn "Regla" docs/
0	directorio sólo con YAML	grep -rn port deploy/dev/

# Comandos que no son búsqueda en absoluto.
0	git status	git status
0	ls	ls -la pkg/
0	instalar el buscador no es buscar	brew install ripgrep
0	preguntar dónde está el buscador	command -v rg

# ── UN PATRÓN NUNCA ES UNA RUTA ──────────────────────────────────────────
#
# El acoplamiento que esto corrige: con una lista fija de directorios, un
# PATRÓN que contenía `services` bloqueaba aunque se buscara en documentación.
0	la palabra services como patrón, sobre docs	grep -rn "services" docs/
0	git grep de la palabra services sobre docs	git grep -n services -- docs/
0	patrón con forma de ruta, sobre docs	grep -rn "src/app.ts" docs/
0	git grep acotado a markdown	git grep -n handler -- "*.md"

# ── SIN RUTA: la búsqueda recorre el cwd, que tiene código ────────────────
2	rg sin ruta	rg -n handler
2	git grep sin ruta	git grep -n handler
2	git -C hacia un directorio con código	git -C pkg grep -n handler
2	git grep sobre una revisión, sin ruta	git grep -n handler HEAD
0	git grep sobre una revisión, acotado a docs	git grep -n handler HEAD -- docs/
0	rg acotado a markdown por tipo	rg -n handler -t md
0	rg acotado a markdown por glob	rg -n handler -g "*.md"
0	grep -r acotado a markdown	grep -rn handler --include="*.md" .
2	rg acotado a código por tipo	rg -n handler -t ts

# ── NO SE PUEDE MEDIR: no abre ─────────────────────────────────────────────
2	directorio en una variable	grep -rn x "$DIR"
0	directorio que no existe (no hay código que leer)	grep -rn x nada/

# ── HEREDOCS: texto escrito vs ejecución ─────────────────────────────────
#
# El cuerpo de un heredoc es TEXTO que se escribe a un archivo (así se escriben
# los mensajes de commit largos). Pero un heredoc dirigido a un INTÉRPRETE sí
# es ejecución. El runner es UNA línea por caso: el hook ve el mismo texto.
0	heredoc de texto que describe búsquedas	cat > /tmp/m.txt <<EOF ; grep -rn x pkg/ ; EOF
2	heredoc a bash ES ejecución	bash <<EOF ; grep -rn x pkg/ ; EOF

# ── Evasiones y falsos positivos ─────────────────────────────────────────
#
# Entrecomillar el directorio no lo esconde. Y una extensión de no-código
# decide aunque el archivo viva en un directorio de código.
2	directorio entrecomillado	grep -rn handler "pkg"
2	find con xargs -0	find pkg -name "*.ts" -print0 | xargs -0 cat
2	find con -exec	find pkg -name "*.ts" -exec cat {} +
2	ugrep recursivo	ugrep -rn handler pkg/
2	sudo no cambia la búsqueda	sudo grep -rn x pkg/
2	xargs con grep sobre código	find pkg -name "*.ts" | xargs grep -n handler
0	log bajo un directorio de código	grep -rn ERROR pkg/logs/app.log
0	YAML bajo deploy/	grep -n port deploy/dev/docker-compose.yml
0	markdown bajo src/	grep -rn foo src/README.md
0	awk sobre salida de CLI	docker compose ps | awk "{print $1}"
0	find que sólo lista	find pkg -name "*.ts"

# ── LO QUE NO ES BUSCAR EN CODIGO ────────────────────────────────────────

# (A) grep sobre la SALIDA DE UN COMANDO. El índice no puede responder
# "cuántos tests fallaron" ni "qué contenedor está healthy".
0	grep sobre la salida de los tests	pnpm test 2>&1 | grep -E "Tests |Test Files |FAIL"
0	grep sobre vitest	npx vitest run 2>&1 | grep -E "^ *Tests |FAIL"
0	git ls-files a grep	git ls-files | grep -i "userprofile"
0	ls a grep	ls db/*.sql | grep -iE "init"

# (B) Buscar por NOMBRE DE ARCHIVO no es buscar EN el archivo.
0	find -name solo lista nombres	find pkg -name "*user-profile*"
0	find con varias extensiones	find deploy -name "*.tf" -o -name "*.tfvars"
0	git ls-files a secas	git ls-files

# (C) Un bucle sobre archivos CONCRETOS no es un recorrido de árbol.
0	for sobre archivos nombrados uno a uno	for f in a.tf b.tf; do grep -c X "$f"; done

# ── Y LO QUE ESAS EXENCIONES NO PUEDEN ABRIR ─────────────────────────────
#
# El pipe no absuelve a un grep que SÍ tiene ruta propia, y listar nombres no
# absuelve a un volcado de contenido.
2	pipe pero el grep recurre un árbol	echo x | grep -rn handler pkg/
2	pipe pero el grep trae --include	echo x | grep -n --include=*.ts x .
2	grep recursivo seguido de otro grep	grep -rn x pkg/ | grep -v test
2	find que lista y LUEGO vuelca	find pkg -name "*.ts" -print0 | xargs -0 cat
2	find con -exec cat	find pkg -name "*.ts" -exec cat {} +
2	ls que lista y LUEGO vuelca	ls pkg/a/src/*.ts | xargs cat
2	ls seguido de un grep recursivo	ls && grep -rn handler pkg/
2	cat por comodín	cat pkg/a/src/*.ts
2	bucle que vuelca un directorio con código	for f in pkg/a/src/*; do cat "$f"; done
