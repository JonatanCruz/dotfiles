# Casos de no-leer-secretos-por-bash.sh
#
# ── QUÉ CIERRA ─────────────────────────────────────────────────────────────
#
# Las 7 reglas `deny` de LECTURA del usuario gobiernan la herramienta `Read`.
# Bash era otra puerta que no las respetaba: `cat secrets/prod.key` daba exit 0
# mientras `Read` lo bloqueaba. Ningún hook del sistema las cubría — todos
# están hechos de verbos de ESCRITURA.
#
# ── EL RIESGO AQUÍ ES BLOQUEAR DE MÁS ──────────────────────────────────────
#
# Las reglas `*password*` y `*credentials*` matchean CÓDIGO FUENTE legítimo
# (`reset-user-password.usecase.ts`, `secret-loader.service.ts`). Si leerlos
# se bloquea, el gate muerde trabajo normal y se termina rodeando. Por eso hay
# tantos casos de exit 0 como de exit 2.
#
# Formato:  <exit esperado> <TAB> <descripción> <TAB> <comando>

# ── DEBE BLOQUEAR: leer el contenido de un secreto ────────────────────────
#
# Las cinco que se midieron pasando, una por regla deny.

2	secrets/ con cat	cat secrets/prod.key
2	.ssh con cat	cat .ssh/id_rsa
2	.aws/credentials	cat .aws/credentials
2	archivo *password*	cat config/db-password.txt
2	.pem con head	head -5 certs/server.pem

# El .env: el archivo de secretos más común de un proyecto, y no estaba en
# NINGUNA regla deny — legible por Read y por Bash.
2	.env con cat	cat .env
2	.env por grep	grep DATABASE_URL .env

# ── Heredoc: texto escrito vs ejecución ─────────────────────────────────
#
# Un `cat > x.md <<EOF` que DOCUMENTA el gate menciona rutas protegidas sin
# abrir ninguna. Bloquearlo muerde el trabajo de escribir SOBRE el gate — el
# caso real fue redactar el cuerpo de un PR que las listaba en una tabla.
# Un heredoc dirigido a un intérprete sí es ejecución y sigue bloqueando.
# Los casos de heredoc REAL (con saltos de línea) viven en
# scripts/tests/hook-secretos.sh: este formato es de una línea, y un heredoc en
# una sola línea no es un heredoc para bash.
2	heredoc a bash ES ejecucion	bash <<EOF ; cat .env ; EOF

# El propio archivo de casos nombra rutas protegidas por definición: es la
# suite del gate. Sin la excepción, el gate impide escribir sus tests.
0	escribir los casos del gate	printf 'x' >> tests/casos-no-leer-secretos-por-bash.txt

# ── Ataque propio al hook, ronda 2 ───────────────────────────────────────
#
# Cuatro evasiones encontradas atacando el diseño desde adentro, por DOS
# causas distintas:
#
#  · GLOBS: el shell expande `.e*` ANTES de ejecutar, así que el hook veía
#    texto sin `.env`. Un comodín no se resuelve sin tocar disco → se toma el
#    lado seguro y se trata como si nombrara la ruta.
#  · SUSTITUCIÓN: `echo $(cat .env)` arrancaba con `echo` (INOCUO) y el lector
#    viajaba dentro de `$(...)`. Misma forma que el bypass `| tee` de la
#    ronda 1: mirar sólo la primera palabra del comando.
2	glob con ?	cat .e?
2	glob con *	cat .e*
2	glob en secrets/	cat secrets*/prod.key
2	lector en sustitución	echo $(cat .env)
2	lector en backticks	echo `cat .env`

# Y lo que NO debe romperse al cerrar lo de arriba: una sustitución sin
# secreto, o con una operación que no expone contenido, sigue siendo legítima.
0	sustitución sin secreto	echo $(git rev-parse HEAD)
# Con un secreto nombrado, una sustitución NUNCA tiene exención (fail-closed):
# distinguir `$(ls …)` de `$(cat …)` es reimplementar el parser de bash, que es
# justo lo que se descartó. Falso positivo asumido.
2	ls dentro de sustitución	echo $(ls secrets/)
0	echo que menciona un template	echo "revisando el .env.example"

# ── EJECUTAR ≠ NOMBRAR: la palabra viaja en TEXTO, no abre nada ───────────
#
# El gate decidia sobre el texto CRUDO del comando, asi que una palabra
# protegida dentro de un PATRON de busqueda, de una REGEX o de un COMENTARIO
# contaba como si fuera una ruta. Ninguno de estos abre un archivo.
#
# El caso testigo es el primero: el hook se bloqueaba a si mismo cuando se
# contaban sus propios bloqueos. Un gate que impide MEDIRLO no se puede
# mantener, y el costo real no es la friccion: es la medicion que no se hizo.
0	el propio nombre del hook como patron	grep -c "no-leer-secretos-por-bash" /tmp/x.jsonl
0	token dentro de una regex de python	python3 -c "import re; re.search(r'name: ' + k + r'value: ', s)"
0	comentario que menciona secretos	ls -la # --- y el de secretos ---
0	patron con punto sobre un .tf	grep -n "local.shared_secrets" deploy/shared/iam.tf
0	patron posicional, archivo despues es .tf	grep -n "shared_secrets" deploy/main.tf
